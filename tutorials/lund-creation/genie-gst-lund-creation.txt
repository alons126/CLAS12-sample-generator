# Run these commands from the repository root to convert physical GENIE GST
# samples into the LUND inputs used by genie-gst-slurm-submission.txt.
#
# Expected input layout for each quoted glob:
#   SAMPLE_DIRECTORY/input_options.txt
#   SAMPLE_DIRECTORY/master-routine_validation_01-eScattering/*.root
#
# The profile uses tune=auto. The converter reads the exact TUNE value from
# input_options.txt and falls back to unknown when discovery fails. The explicit
# target and beam energy select the normal GEMC target variation and vertex geometry.
# Replace every /path/to/... placeholder before running these commands.

#########################################################################################
# Physical C12 GENIE GST conversion at 2.07052 GeV
# Automatic target variation: rgm_fall2021_C_S (small one-foil target)
#########################################################################################

source run.csh \
  --workflow create-lund \
  --source physical \
  --config config/samples/physical-lund-creation/genie-gst.conf \
  --event-generator genie-gst \
  --output-layout nested \
  --target C12 \
  --beam-energy 2.07052 \
  --q2-cut Q2_0.02 \
  --input '/path/to/C12/2070MeV_Q2_0.02/master-routine_validation_01-eScattering/*.root' \
  --events 1000000 \
  --output /path/to/physical-genie-gst-output
