# Physical LUND converter option inventory
#
# Together these commands show every public physical-conversion option. The converter
# reads existing GENIE GST truth; it does not run GENIE. Arbitrary values demonstrate
# syntax, not production settings. Quote input globs so ROOT receives them unchanged,
# replace every /path/to/... placeholder, and commit changes before using run.csh on
# the disposable ifarm checkout. Use the built converter for local development.

#########################################################################################
# Print the current physical LUND converter interface
#########################################################################################

source run.csh \
    --workflow create-lund \
    --source physical \
    --build false \
    -- \
    --help

#########################################################################################
# Arbitrary metadata-layout study using every physical and common option
#
# seed is accepted and recorded by the shared configuration layer but is unused by
# physical conversion because GST supplies particle kinematics. vertex-seed controls
# the independently sampled target position.
#########################################################################################

source run.csh \
    --workflow create-lund \
    --source physical \
    --config config/samples/physical-lund-creation/genie-gst.conf \
    --input '/path/to/arbitrary-gst-study/*.root' \
    --event-generator genie-gst \
    --event-generator-version 3.8.1 \
    --tune GTEST_42 \
    --q2-cut study-Q2-0.17 \
    --output-layout metadata \
    --beam-energy 3.75 \
    --target Ar40 \
    --gemc-target-variation rgm_fall2021_Ar \
    --A 40 \
    --Z 18 \
    --events 317 \
    --events-per-file 83 \
    --seed 11223 \
    --vertex-seed 44556 \
    --prefix arbitrary-genie-gst \
    --output /path/to/arbitrary-physical-output
