# needed to reach it. Arbitrary values demonstrate syntax and validation; they are

# launcher on ifarm, or use the built executable locally. Before starting, run.csh
# discards uncommitted edits and deletes untracked and ignored files except build/,
# then updates from Git. Commit and push development changes from your local copy
# first. Even forwarding application help runs after this cleanup.

#########################################################################################
# Print the current uniform LUND creator interface
#########################################################################################

source run.csh \
    --workflow create-lund \
    --source uniform \
    --build false \
    -- \
    --help

#########################################################################################
# Arbitrary 1e study: all common options and all electron-range options
#
# The profile is read first. Every explicit option below overrides its matching value.
# Ar40 permits an arbitrary positive beam energy because its detector variation and
# vertex geometry do not depend on one of the three standard RG-M beam values.
#########################################################################################

source run.csh \
    --workflow create-lund \
    --source uniform \
    --config config/samples/uniform-lund-creation/uniform-1e-5986MeV.conf \
    --channel 1e \
    --beam-energy 3.75 \
    --target Ar40 \
    --gemc-target-variation rgm_fall2021_Ar \
    --A 40 \
    --Z 18 \
    --events 137 \
    --events-per-file 41 \
    --seed 24680 \
    --vertex-seed 13579 \
    --prefix arbitrary-electrons \
    --electron-theta-min 7.5 \
    --electron-theta-max 32.5 \
    --electron-momentum uniform \
    --electron-p-min 0.85 \
    --electron-p-max 3.25 \
    --output /path/to/arbitrary-uniform-output

#########################################################################################
# Arbitrary electron-hadron study: every sampled-hadron and trigger option
#
# hadron-p remains a checked option but is consumed only when hadron-momentum=fixed.
# The following mixed charged-pion study uses hadron-p-min instead.
#########################################################################################

source run.csh \
    --workflow create-lund \
    --source uniform \
    --channel eh \
    --hadron pim \
    --hadron-region CD \
    --beam-energy 4.2 \
    --target Ar40 \
    --gemc-target-variation rgm_fall2021_Ar \
    --A 40 \
    --Z 18 \
    --events 211 \
    --events-per-file 73 \
    --seed 86420 \
    --vertex-seed 97531 \
    --prefix arbitrary-epimCD \
    --electron-theta-min 5 \
    --electron-theta-max 40 \
    --electron-momentum beam \
    --electron-p-min 0.7 \
    --electron-p-max 4.2 \
    --hadron-theta-min 42 \
    --hadron-theta-max 133 \
    --hadron-momentum mixed \
    --hadron-p-min 0.16 \
    --hadron-p 1.15 \
    --trigger-theta 27 \
    --trigger-phi-offset 9 \
    --output /path/to/arbitrary-uniform-output

#########################################################################################
# Fixed neutron momentum: demonstrate the mode that consumes --hadron-p
#########################################################################################

source run.csh \
    --workflow create-lund \

    --hadron-region FD \
