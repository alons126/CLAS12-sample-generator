# Launcher option examples
#
# Run these commands from the repository root. The launcher options control
# configuring, building, and starting a LUND application. They do not configure
# detector submission. Replace /path/to/... values before running an example.

#########################################################################################
# Print launcher help without updating, building, creating output, or submitting jobs
#########################################################################################

source run.csh \
    --help

#########################################################################################
# Use every launcher build/run option with arbitrary development values
#########################################################################################

source run.csh \
    --run-settings config/run.json \
    --workflow create-lund \
    --source uniform \
    --build true \
    --run false \
    --build-dir build/tutorial-relwithdebinfo \
    --build-type RelWithDebInfo \
    --jobs 3

#########################################################################################
# Run an existing build without configuring or compiling it again
#########################################################################################

source run.csh \
    --workflow create-lund \
    --source uniform \
    --build false \
    --run true \
    --build-dir build/release \
    --config config/samples/uniform-lund-creation/uniform-1e-5986MeV.conf \
    --events 25 \
    --output /path/to/tutorial-output

#########################################################################################
# Forward --help to one selected LUND executable
#
# The standalone -- token ends launcher-option parsing. It is useful when a child
# option could otherwise be mistaken for a launcher option.
#########################################################################################

source run.csh \
    --workflow create-lund \
    --source uniform \
    --build false \
    -- \
    --help

source run.csh \
    --workflow create-lund \
    --source physical \
    --build false \
    -- \
    --help

#########################################################################################
# Select the submission workflow
#
# Submission has no --source and bypasses the launcher build controls. It previews
# by default. Its complete option inventory is in slurm-submission/all-options.txt.
#########################################################################################

source run.csh \
    --workflow submit \
    --lund-dir /path/to/completed-run/lundfiles
